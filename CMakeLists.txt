cmake_minimum_required(VERSION 3.18)
project(pcicf_cpp VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

find_package(nlohmann_json 3.10 CONFIG REQUIRED)
find_package(yaml-cpp CONFIG REQUIRED)
find_package(Eigen3 3.3 CONFIG REQUIRED)
# Require the version verified with the supplied YOLO11x ONNX model.
# Select an alternative installation explicitly with OpenCV_DIR (see README).
find_package(OpenCV 4.10.0 EXACT REQUIRED CONFIG
             COMPONENTS core imgproc imgcodecs dnn video calib3d)
message(STATUS "PCICF OpenCV ${OpenCV_VERSION}: ${OpenCV_DIR}")
if(TARGET yaml-cpp::yaml-cpp)
    set(YAML_TARGET yaml-cpp::yaml-cpp)
else()
    set(YAML_TARGET yaml-cpp)
endif()

add_library(pcicf_core src/core.cpp src/tracker.cpp src/detector.cpp src/plot.cpp)
target_include_directories(pcicf_core PUBLIC include ${OpenCV_INCLUDE_DIRS})
target_link_libraries(pcicf_core PUBLIC nlohmann_json::nlohmann_json ${YAML_TARGET} Eigen3::Eigen ${OpenCV_LIBS})
add_executable(detection_pipeline src/detection_pipeline.cpp)
target_link_libraries(detection_pipeline PRIVATE pcicf_core)
if(MSVC)
    target_compile_options(pcicf_core PRIVATE /W4)
    target_compile_options(detection_pipeline PRIVATE /W4)
else()
    target_compile_options(pcicf_core PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(detection_pipeline PRIVATE -Wall -Wextra -Wpedantic)
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(core_checks tests/core_checks.cpp)
    target_link_libraries(core_checks PRIVATE pcicf_core)
    add_test(NAME core_checks COMMAND core_checks)
endif()
